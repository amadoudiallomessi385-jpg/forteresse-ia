async function envoyer() { const input = document.getElementById('text'); const message = input.value; if(!message) return; add(message, 'user'); input.value = ""; try { // On appelle le Gardien (api/chat.js) que tu as créé const response = await fetch('/api/chat', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ message: message }) }); const data = await response.json(); if (data.reply) { add(data.reply, 'ai'); } else { add("Le Gardien n'a pas pu obtenir de réponse.", 'ai'); } } catch (e) { add("La liaison avec le Gardien a échoué.", 'ai'); } }
